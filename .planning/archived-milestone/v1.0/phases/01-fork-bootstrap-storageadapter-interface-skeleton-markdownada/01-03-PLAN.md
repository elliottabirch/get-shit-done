    ```typescript
    export class MarkdownAdapter implements StorageAdapter {
      readonly name = 'markdown';
      readonly capabilities: Capabilities = {
        record: true,
        section: true,
        frontmatter: true,
        binaryAsset: false,
        snapshot: false,
        transaction: false,
        namedDoc: false,
        commitPlanningState: true,
        markdownLockfile: true,
      };

      private readonly projectDir: string;
      private readonly planningBase: string;
      private readonly req: NodeRequire;
      private readonly libPaths: { [name: string]: string };

      constructor(projectDir: string) {
        this.projectDir = projectDir;
        this.req = createRequire(import.meta.url);
        // Pre-resolve every CJS path the adapter will need
        this.libPaths = {
          'core.cjs': resolveLibPath('core.cjs', projectDir),
          'frontmatter.cjs': resolveLibPath('frontmatter.cjs', projectDir),
          'planning-workspace.cjs': resolveLibPath('planning-workspace.cjs', projectDir),
        };
        const ws = this.req(this.libPaths['planning-workspace.cjs']) as {
          planningDir: (cwd: string) => string;
        };
        this.planningBase = ws.planningDir(projectDir);
      }

      private resolve(relPath: string): string {
        return join(this.planningBase, relPath);
      }

      // ─── Bin A: record group ───────────────────────────────────────────────

      async getRecord(path: string): Promise<string | null> {
        const abs = this.resolve(path);
        try {
          return await readFile(abs, 'utf-8');
        } catch (err) {
          if ((err as NodeJS.ErrnoException).code === 'ENOENT') return null;
          throw err;
        }
      }

      async putRecord(path: string, body: string): Promise<void> {
        const abs = this.resolve(path);
        await mkdir(dirname(abs), { recursive: true });
        await writeFile(abs, body, 'utf-8');
      }

      async removeRecord(path: string): Promise<void> {
        const abs = this.resolve(path);
        try {
          await unlink(abs);
        } catch (err) {
          if ((err as NodeJS.ErrnoException).code === 'ENOENT') return;
          throw err;
        }
      }

      async listCollection(prefix: string, filter?: RecordFilter): Promise<RecordRef[]> {
        const abs = this.resolve(prefix);
        let entries: import('node:fs').Dirent[];
        try {
          entries = await readdir(abs, { withFileTypes: true });
        } catch (err) {
          if ((err as NodeJS.ErrnoException).code === 'ENOENT') return [];
          throw err;
        }
        const refs: RecordRef[] = entries.map(e => ({
          path: join(prefix, e.name),
          name: e.name,
        }));
        return filter ? refs.filter(filter) : refs;
      }

      async exists(path: string): Promise<boolean> {
        return existsSync(this.resolve(path));
      }

      // ─── Bin A: section group (regex impl — no CJS analog per RESEARCH.md Pitfall 5) ───

      async getSection(path: string, anchor: string): Promise<string | null> {
        const text = await this.getRecord(path);
        if (text === null) return null;
        // Heading-level-2 anchor match: ## anchor, until next ## or EOF
        const escaped = anchor.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        const re = new RegExp(`^##\\s+${escaped}\\s*$([\\s\\S]*?)(?=^##\\s|\\Z)`, 'm');
        const m = text.match(re);
        return m ? m[1].replace(/^\n+|\n+$/g, '') : null;
      }

      async updateSection(path: string, anchor: string, body: string, mode: SectionMode): Promise<void> {
        const current = (await this.getRecord(path)) ?? '';
        const escaped = anchor.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        const sectionRe = new RegExp(`(^##\\s+${escaped}\\s*$)([\\s\\S]*?)(?=^##\\s|\\Z)`, 'm');
        const match = current.match(sectionRe);

        let next: string;
        if (!match) {
          // Section does not exist — append a new one at the end
          const sep = current.endsWith('\n') ? '' : '\n';
          next = `${current}${sep}\n## ${anchor}\n\n${body}\n`;
        } else {
          const heading = match[1];
          const existing = match[2].replace(/^\n+|\n+$/g, '');
          let newBlock: string;
          switch (mode) {
            case 'overwrite':
              newBlock = `${heading}\n\n${body}\n\n`;
              break;
            case 'append':
              newBlock = `${heading}\n\n${existing}\n\n${body}\n\n`;
              break;
            case 'prepend':
              newBlock = `${heading}\n\n${body}\n\n${existing}\n\n`;
              break;
          }
          next = current.replace(sectionRe, newBlock);
        }
        await this.putRecord(path, next);
      }

      // ─── Bin A: frontmatter group (wrap frontmatter.cjs) ───────────────────

      async getFrontmatter(path: string, field?: string): Promise<unknown> {
        const fm = this.req(this.libPaths['frontmatter.cjs']) as {
          cmdFrontmatterGet: (cwd: string, filePath: string, field?: string, json?: boolean) => unknown;
        };
        return fm.cmdFrontmatterGet(this.projectDir, this.resolve(path), field, false);
      }

      async updateFrontmatter(path: string, field: string, value: unknown): Promise<void> {
        const fm = this.req(this.libPaths['frontmatter.cjs']) as {
          cmdFrontmatterSet: (cwd: string, filePath: string, field: string, value: unknown, json?: boolean) => void;
        };
        fm.cmdFrontmatterSet(this.projectDir, this.resolve(path), field, value, false);
      }

      async mergeFrontmatter(path: string, patch: Record<string, unknown>): Promise<void> {
        const fm = this.req(this.libPaths['frontmatter.cjs']) as {
          cmdFrontmatterMerge: (cwd: string, filePath: string, patch: Record<string, unknown>, json?: boolean) => void;
        };
        fm.cmdFrontmatterMerge(this.projectDir, this.resolve(path), patch, false);
      }

      // ─── markdownLockfile group (D-09 / OQ-08 — implemented since cap=true) ─

      async replaceInCurrentMilestone(pattern: string | RegExp, replacement: string): Promise<void> {
        const core = this.req(this.libPaths['core.cjs']) as {
          replaceInCurrentMilestone: (content: string, pattern: string | RegExp, replacement: string) => string;
          atomicWriteFileSync: (filePath: string, content: string, encoding?: string) => void;
        };
        const roadmapPath = this.resolve('ROADMAP.md');
        const current = (await this.getRecord('ROADMAP.md')) ?? '';
        const updated = core.replaceInCurrentMilestone(current, pattern, replacement);
        core.atomicWriteFileSync(roadmapPath, updated, 'utf-8');
      }

      async readModifyWriteRoadmapMd(mutator: (content: string) => string): Promise<void> {
        const core = this.req(this.libPaths['core.cjs']) as {
          atomicWriteFileSync: (filePath: string, content: string, encoding?: string) => void;
        };
        const roadmapPath = this.resolve('ROADMAP.md');
        const current = (await this.getRecord('ROADMAP.md')) ?? '';
        const updated = mutator(current);
        core.atomicWriteFileSync(roadmapPath, updated, 'utf-8');
      }

      // ─── commitPlanningState (cap=true — implemented in Phase 1 per D-10) ───

      async commitPlanningState(message: string, files?: string[]): Promise<void> {
        // Use git CLI directly (matches existing commit handler behavior)
        const { execFileSync } = await import('node:child_process');
        const args = files && files.length > 0 ? files : ['.planning'];
        try {
          execFileSync('git', ['add', ...args], { cwd: this.projectDir, stdio: 'pipe' });
          execFileSync('git', ['commit', '-m', message], { cwd: this.projectDir, stdio: 'pipe' });
        } catch (err) {
          // If nothing-to-commit, surface a typed error consumers can handle
          throw new Error(`commitPlanningState failed: ${(err as Error).message}`);
        }
      }

      // ─── Foundational primitives — defensive throws (D-10 + D-11) ─────────

      async writeBinaryAsset(_path: string, _bytes: Uint8Array): Promise<void> {
        throw new UnsupportedCapabilityError('binaryAsset', this.name);
      }

      async snapshot(): Promise<string> {
        throw new UnsupportedCapabilityError('snapshot', this.name);
      }

      async restore(_snapshotId: string): Promise<void> {
        throw new UnsupportedCapabilityError('snapshot', this.name);
      }

      async withTransaction(_fn: () => Promise<void>): Promise<void> {
        throw new UnsupportedCapabilityError('transaction', this.name);
      }

      async putNamedDoc(_category: string, _key: string, _body: string): Promise<void> {
        throw new UnsupportedCapabilityError('namedDoc', this.name);
      }

      async getNamedDoc(_category: string, _key: string): Promise<string | null> {
        throw new UnsupportedCapabilityError('namedDoc', this.name);
      }
    }
    ```
