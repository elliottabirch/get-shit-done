    ```ts
    #!/usr/bin/env node
    /**
     * D-MAPPING-SPIKE + D-TXN-SPIKE evidence gatherer for Plan 06-03.
     *
     * Produces SPIKE-RESULTS.md §§1-9 at
     * /Volumes/code/get-shit-done/.planning/phases/06-beadsadapter-implementation/06-03-SPIKE-RESULTS.md
     *
     * Usage:
     *   npx tsx scripts/spike-bd-primitives.ts
     *   # or after `npm run build`:
     *   node dist/scripts/spike-bd-primitives.js
     *
     * Requires: bd v1.0.3 on PATH (fails fast if absent).
     */
    import { spawnSync } from 'node:child_process';
    import { mkdtempSync, writeFileSync, rmSync, mkdirSync } from 'node:fs';
    import { tmpdir } from 'node:os';
    import { join, resolve } from 'node:path';
    import { BdRunner } from '../src/bd/helper.js';
    import { BeadsEmpty, BeadsUnavailableError } from '../src/bd/errors.js';

    const OUT_PATH = resolve(
      '/Volumes/code/get-shit-done/.planning/phases/06-beadsadapter-implementation/06-03-SPIKE-RESULTS.md',
    );

    interface ProbeResult {
      section: string;
      question: string;
      verdict: 'PASS' | 'PARTIAL' | 'FAIL' | 'INCONCLUSIVE';
      command?: string;
      output?: string;
      notes?: string;
    }

    const results: ProbeResult[] = [];
    const version = { raw: '', family: '' };

    // ---------- Pre-flight ----------
    function preflight(): void {
      const which = spawnSync('which', ['bd'], { encoding: 'utf-8' });
      if (which.status !== 0) {
        console.error('FAIL: bd CLI not on PATH. Install bd v1.0.3 before running spike.');
        process.exit(1);
      }
      const v = spawnSync('bd', ['--version'], { encoding: 'utf-8' });
      version.raw = (v.stdout || '').trim();
      if (!/v?1\.0\.3/.test(version.raw)) {
        console.warn(`WARN: bd --version = "${version.raw}" — expected v1.0.3 family. Continuing with caveat.`);
      }
      version.family = /1\.0\.\d+/.exec(version.raw)?.[0] ?? 'unknown';
    }

    // ---------- Fixture ----------
    function setupFixture(): { dir: string; bd: BdRunner } {
      const dir = mkdtempSync(join(tmpdir(), 'spike-bd-'));
      mkdirSync(join(dir, '.planning'), { recursive: true });

      // bd init
      spawnSync('git', ['init', '-q'], { cwd: dir });
      const init = spawnSync('bd', [
        'init', '--non-interactive', '--skip-agents', '--skip-hooks', '--quiet',
      ], { cwd: dir, env: { ...process.env, BEADS_ACTOR: 'seed' }, encoding: 'utf-8' });
      if (init.status !== 0) {
        console.error('FAIL: bd init failed:', init.stderr);
        rmSync(dir, { recursive: true, force: true });
        process.exit(1);
      }
      const bd = new BdRunner(dir);

      // Seed a few issues for probes
      spawnSync('bd', ['create', 'probe-issue-1', '-d', 'seed body 1'], {
        cwd: dir, env: { ...process.env, BEADS_ACTOR: 'seed' },
      });

      return { dir, bd };
    }

    // ---------- Probes ----------
    function probe_namedJsonField(bd: BdRunner): void {
      // §2: Does `bd update <id>` accept arbitrary JSON in a named field beyond --description?
      const list = bd.run(['list', '--json', '--all']) as Array<{ id: string }>;
      const id = list[0]?.id;
      if (!id) {
        results.push({
          section: '§2.1',
          question: 'bd update --json-field / sub-record primitive',
          verdict: 'INCONCLUSIVE',
          notes: 'no seeded issue available',
        });
        return;
      }

      // Try various primitive shapes bd might expose
      const attempts: Array<{ args: string[]; label: string }> = [
        { args: ['update', id, '--json-field', 'custom', '{"key":"value"}'], label: 'bd update --json-field' },
        { args: ['update', id, '--set', 'custom=value'], label: 'bd update --set' },
        { args: ['update', id, '--metadata', '{"custom":"value"}'], label: 'bd update --metadata' },
      ];

      let found = false;
      const outcomes: string[] = [];
      for (const { args, label } of attempts) {
        const r = spawnSync('bd', args, { cwd: process.cwd(), env: { ...process.env, BEADS_ACTOR: 'seed' }, encoding: 'utf-8' });
        const emitted = `exit=${r.status}; stdout=${(r.stdout || '').trim().slice(0,80)}; stderr=${(r.stderr || '').trim().slice(0,80)}`;
        outcomes.push(`${label}: ${emitted}`);
        if (r.status === 0) found = true;
      }
      results.push({
        section: '§2.1',
        question: 'Does bd update accept a named field beyond --description?',
        verdict: found ? 'PASS' : 'FAIL',
        command: attempts.map(a => 'bd ' + a.args.join(' ')).join(' | '),
        output: outcomes.join('\n'),
        notes: found ? 'at least one named-field primitive succeeded — investigate round-trip' : 'sibling empirical finding confirmed: no named-JSON-field primitive in bd v1.0.3',
      });

      // Additionally check `bd update --help` for any --* flags that carry JSON
      const help = spawnSync('bd', ['update', '--help'], { encoding: 'utf-8' });
      results.push({
        section: '§2.2',
        question: 'bd update --help flag inventory',
        verdict: 'PASS',
        command: 'bd update --help',
        output: (help.stdout || '').slice(0, 600),
      });
    }

        verdict: isJsonl ? 'PASS' : (body.startsWith('[') ? 'FAIL' : 'INCONCLUSIVE'),
    ```
