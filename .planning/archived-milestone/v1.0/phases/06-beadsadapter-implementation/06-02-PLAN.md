    /**
     * Sentinel errors for bd CLI invocation + BeadsAdapter lifecycle.
     * Ported from sibling src/bd/errors.mjs; UnsupportedOperationError REMOVED
     * (per CONTEXT.md D-BINARY — consumers import UnsupportedCapabilityError
     * from `get-shit-done/adapters/types.js`).
     * BdManagedMismatchError ADDED per D-INIT-ERR (BEADS-04).
     */

    export const BeadsCause = Object.freeze({
      NotInstalled: 'not-installed',
      Corrupt: 'corrupt',
      VersionMismatch: 'version-mismatch',
      Empty: 'empty',
      Unknown: 'unknown',
      Unsupported: 'unsupported',
    } as const);
    export type BeadsCauseValue = typeof BeadsCause[keyof typeof BeadsCause];

    export class BeadsUnavailableError extends Error {
      override readonly name: string = 'BeadsUnavailableError';
      readonly cause: BeadsCauseValue;
      readonly originalError?: Error;

      constructor(message: string, opts: { cause?: BeadsCauseValue; originalError?: Error } = {}) {
        super(message);
        this.cause = opts.cause ?? BeadsCause.Unknown;
        if (opts.originalError !== undefined) this.originalError = opts.originalError;
      }
    }

    export class BeadsNotInstalled extends BeadsUnavailableError {
      override readonly name = 'BeadsNotInstalled';
      constructor(message = 'bd CLI not found on PATH', originalError?: Error) {
        super(message, { cause: BeadsCause.NotInstalled, originalError });
      }
    }

    export class BeadsCorrupt extends BeadsUnavailableError {
      override readonly name = 'BeadsCorrupt';
      constructor(message: string, originalError?: Error) {
        super(message, { cause: BeadsCause.Corrupt, originalError });
      }
    }

    export class BeadsVersionMismatch extends BeadsUnavailableError {
      override readonly name = 'BeadsVersionMismatch';
      constructor(message: string, originalError?: Error) {
        super(message, { cause: BeadsCause.VersionMismatch, originalError });
      }
    }

    export class BeadsEmpty extends BeadsUnavailableError {
      override readonly name = 'BeadsEmpty';
      constructor(message = 'bd store is empty', originalError?: Error) {
        super(message, { cause: BeadsCause.Empty, originalError });
      }
    }

    /**
     * D-INIT-ERR (BEADS-04): typed error thrown by BeadsAdapter when the
     * target project directory is not bd-managed. Shape mirrors fork's
     * UnsupportedCapabilityError (__brand symbol enables cross-module
     * instanceof resilience under vitest transforms / dual-package hazards).
     *
     * Fields:
     *   - code: 'PROJECT_BD_MANAGED_MISMATCH' (literal; machine-readable)
     *   - projectDir: path that failed the probe
     *   - hint: human-readable next-step guidance
     */
    export class BdManagedMismatchError extends Error {
      override readonly name = 'BdManagedMismatchError';
      readonly code = 'PROJECT_BD_MANAGED_MISMATCH' as const;
      readonly projectDir: string;
      readonly hint: string;
      readonly __brand = 'BdManagedMismatchError' as const;

      constructor(projectDir: string, hint: string) {
        super(`Project directory '${projectDir}' is not bd-managed. ${hint}`);
        this.projectDir = projectDir;
        this.hint = hint;
      }

      static [Symbol.hasInstance](instance: unknown): boolean {
        return (
          instance != null &&
          typeof instance === 'object' &&
          (instance as Record<string, unknown>).__brand === 'BdManagedMismatchError'
        );
      }
    }
    ```

    import { existsSync, realpathSync } from 'node:fs';
    import { join, dirname, resolve } from 'node:path';

    /**
     * Locate the nearest .beads/metadata.json by walking upward from startDir.
     *
     * 4 topology cases (ported from sibling; spike-003 VALIDATED):
     *   1. BEADS_DIR env override (explicit bd root path)
     *   2. Parent-walk for .beads/metadata.json
     *   3. .git-as-file (worktree — stops walk at primary repo boundary without inventing fallback)
     *   4. No bd anywhere → null
     *
     * Returns the absolute path of the .beads directory, or null.
     */
    export function findBeadsRoot(startDir: string): string | null {
      // Case 1: BEADS_DIR env override
      if (process.env.BEADS_DIR) {
        const candidate = resolve(process.env.BEADS_DIR);
        if (existsSync(join(candidate, 'metadata.json'))) return candidate;
        return null;   // env set but invalid — do NOT fall through
      }

      // Cases 2-4: parent-walk
      let dir: string;
      try {
        dir = realpathSync(startDir);
      } catch {
        return null;
      }

      while (true) {
        const beadsDir = join(dir, '.beads');
        if (existsSync(join(beadsDir, 'metadata.json'))) return beadsDir;

        const parent = dirname(dir);
        if (parent === dir) break;    // filesystem root
        dir = parent;
      }
      return null;
    }

    import { spawnSync } from 'node:child_process';
    import {
      BeadsNotInstalled,
      BeadsCorrupt,
      BeadsEmpty,
      BeadsUnavailableError,
      BeadsCause,
    } from './errors.js';

    /**
     * Adapter-context-bound bd CLI wrapper. Fixes sibling's landmines 3 + 4 + 11
     * at the wrapper layer so individual call sites in primitives.ts /
     * events.ts / dep-graph.ts don't have to repeat the discipline.
     *
     * Landmine 3 fix: cwd baked at construction (per adapter instance).
     * Landmine 4 fix: env forwarded to spawnSync (sibling's bd() wrapper did NOT).
     * Landmine 11 fix: BEADS_ACTOR=seed default (CONF-03 byte-identity).
     * Landmine 5: show() unwraps single-element array from `bd show <id> --json`.
     * Landmine 6: JSONL fallback when stdout isn't valid JSON array (bd export --json).
     * Landmine 7: {error, schema_version} at exit 0 → BeadsEmpty sentinel.
     */
    export class BdRunner {
      constructor(
        private readonly cwd: string,
        private readonly baseEnv: NodeJS.ProcessEnv = { ...process.env, BEADS_ACTOR: 'seed' },
      ) {}

      /**
       * Invoke bd with args; parse stdout as JSON (or JSONL fallback).
       * Throws typed sentinels (BeadsNotInstalled / BeadsCorrupt / BeadsEmpty / BeadsUnavailableError)
       * on failure.
       */
      run(args: string[], opts?: { parseJson?: boolean; env?: Record<string, string> }): unknown {
        const parseJson = opts?.parseJson ?? true;
        const env: NodeJS.ProcessEnv = { ...this.baseEnv, ...(opts?.env ?? {}) };

        const result = spawnSync('bd', args, { cwd: this.cwd, env, encoding: 'utf-8' });

        // ENOENT: bd not installed
        if (result.error && (result.error as NodeJS.ErrnoException).code === 'ENOENT') {
          throw new BeadsNotInstalled('bd CLI not found on PATH', result.error);
        }

        // Non-zero exit or other spawn error
        if (result.error || (typeof result.status === 'number' && result.status !== 0)) {
          const stderr = (result.stderr || '').toString();
          if (/database is locked|schema mismatch|corrupt/i.test(stderr)) {
            throw new BeadsCorrupt(`bd failure: ${stderr.slice(0, 200)}`, result.error ?? undefined);
          }
          throw new BeadsUnavailableError(
            `bd failed (exit ${result.status}): ${stderr.slice(0, 200)}`,
            { cause: BeadsCause.Unknown, originalError: result.error ?? undefined },
          );
        }

        if (!parseJson) return (result.stdout || '').toString().trim() || null;

        const stdoutText = (result.stdout || '').toString().trim();
        if (stdoutText.length === 0) return null;

        // Landmine 7: {error, schema_version} empty-store shape at exit 0
        try {
          const parsed = JSON.parse(stdoutText);
          if (
            parsed !== null &&
            typeof parsed === 'object' &&
            'error' in parsed &&
            'schema_version' in parsed
          ) {
            throw new BeadsEmpty(`bd reports empty store: ${(parsed as { error: string }).error}`);
          }
          return parsed;
        } catch (e) {
          if (e instanceof BeadsEmpty) throw e;
          // Landmine 6: JSONL fallback for `bd export --json`
          try {
            const lines = stdoutText.split('\n').filter((l) => l.trim().length > 0);
            return lines.map((l) => JSON.parse(l));
          } catch {
            throw new BeadsCorrupt(
              `bd stdout is neither JSON nor JSONL: ${stdoutText.slice(0, 200)}`,
            );
          }
        }
      }

      /**
       * Landmine 5 fix: `bd show <id> --json` returns single-element array in v1.0.3.
       * Always unwrap.
       */
      show(id: string): Record<string, unknown> {
        const r = this.run(['show', id, '--json']);
        if (Array.isArray(r)) {
          if (r.length === 0) throw new BeadsEmpty(`bd show ${id}: no result`);
          return r[0] as Record<string, unknown>;
        }
        return r as Record<string, unknown>;
      }
    }

    import { writeFileSync, renameSync, mkdirSync } from 'node:fs';
    import { resolve, dirname, basename } from 'node:path';
    import { randomBytes } from 'node:crypto';

    /**
     * Atomic file write via tmpfile + POSIX rename in the same directory.
     *
     * Landmine 15 / WR-05 fix: sibling used `${pid}.${Date.now()}` which collides
     * under ms-resolution concurrent writers. Swapped to `crypto.randomBytes(6)`
     * hex suffix (48 bits of entropy) — effectively zero collision probability
     * across realistic concurrency.
     *
     * Same-dir rename is load-bearing for POSIX atomicity guarantee — do NOT
     * move tmpfile to /tmp or equivalent.
     */
    export function atomicWriteFile(absPath: string, body: string | Buffer): void {
      const dir = dirname(absPath);
      mkdirSync(dir, { recursive: true });
      const suffix = `${process.pid}.${randomBytes(6).toString('hex')}`;
      const tmpPath = resolve(dir, `.${basename(absPath)}.tmp.${suffix}`);
      writeFileSync(tmpPath, body);
      renameSync(tmpPath, absPath);
    }
    ```
