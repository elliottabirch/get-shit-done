```typescript
import type { BdClient } from '../bd-client.js';
import type { GraphEdge } from './graph-schema.js';

const PATH_LABEL_PREFIX = 'gsd:path:';

function extractPath(labels: string[]): string | null {
  const label = labels.find(l => l.startsWith(PATH_LABEL_PREFIX));
  return label ? label.slice(PATH_LABEL_PREFIX.length) : null;
}

export async function synthesizeDepEdges(bdClient: BdClient): Promise<GraphEdge[]> {
  const rawEdges = await bdClient.issueListBlocksEdges();
  if (rawEdges.length === 0) return [];

  // Batch-resolve ids to paths. Simple impl: per-id fetch; cache repeats.
  const idToPath = new Map<string, string | null>();
  async function resolve(issueId: string): Promise<string | null> {
    if (idToPath.has(issueId)) return idToPath.get(issueId)!;
    const issue = await bdClient.issueGetById(issueId);
    const path = issue ? extractPath(issue.labels) : null;
    idToPath.set(issueId, path);
    return path;
  }

  const depEdges: GraphEdge[] = [];
  for (const e of rawEdges) {
    const fromPath = await resolve(e.fromIssueId);
    const toPath = await resolve(e.toIssueId);
    if (fromPath === null || toPath === null) continue;  // orphan edge — skip
    depEdges.push({
      type: 'dependency',
      confidence: 1.0,
      from: fromPath,
      to: toPath,
    });
  }
  return depEdges;
}
```

async function synthesizeGraphJson(adapter: TransactionalAdapter, path: string): Promise<string> {
  // 1. Read stored body via the normal getRecord path (temporarily bypass the synthesizer).
  //    Implementation trick: inline the Plan-02 body-read logic here, not a recursive call.
  const storedBody = await readStoredBody(adapter, path);  // helper that does Plan 02 logic sans synthesis
  const stored: Graph = storedBody !== null ? JSON.parse(storedBody) : emptyGraph();

  // 2. Synthesize bd dep edges
  const depEdges = await synthesizeDepEdges(adapter.bdClient);

  // 3. Merge + dedupe by (type, from, to)
  const seen = new Set(stored.edges.map(edgeKey));
  const merged = [...stored.edges];
  for (const e of depEdges) {
    if (!seen.has(edgeKey(e))) {
      merged.push(e);
      seen.add(edgeKey(e));
    }
  }

  const outGraph: Graph = { ...stored, nodes: stored.nodes, edges: merged };
  return JSON.stringify(outGraph);
}
