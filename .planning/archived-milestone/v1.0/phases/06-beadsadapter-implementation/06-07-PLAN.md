The BeadsAdapter factory therefore performs bd init inline:

```ts
// tests/conformance.test.ts (target shape)
import { runAdapterConformanceSuite } from 'get-shit-done/conformance';
import { spawnSync } from 'node:child_process';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { chmodSync, copyFileSync, mkdirSync } from 'node:fs';
import { BeadsAdapter } from '../src/index.js';

const __dirname = dirname(fileURLToPath(import.meta.url));
const SEED_PATH = join(__dirname, 'fixtures/seed.jsonl');

runAdapterConformanceSuite('beads', (projectDir: string) => {
  // Harness pre-created projectDir + .planning. We must bd init here.
  // BEADS_ACTOR=seed for byte-identity discipline (Landmine 11).
  // git init first — bd init pre-checks for a git repo in v1.0.3.
  spawnSync('git', ['init', '-q'], { cwd: projectDir, stdio: 'ignore' });
  spawnSync('git', ['config', 'user.email', 'seed@test.local'], { cwd: projectDir });
  spawnSync('git', ['config', 'user.name', 'Seed Test'], { cwd: projectDir });

  const init = spawnSync('bd', [
    'init', '--from-jsonl', SEED_PATH,
    '--non-interactive', '--skip-agents', '--skip-hooks', '--quiet',
  ], {
    cwd: projectDir,
    env: { ...process.env, BEADS_ACTOR: 'seed' },
    encoding: 'utf-8',
  });
  if (init.status !== 0) {
    throw new Error(`bd init failed in conformance factory: ${init.stderr}`);
  }
  // Landmine 9: .beads chmod 0o700
  try { chmodSync(join(projectDir, '.beads'), 0o700); } catch { /* non-fatal */ }

  return new BeadsAdapter(projectDir);
});
```
