Create `sdk/src/query/verify-fat-skills.ts` implementing the `verify.fat-skills` SDK query per D-08.

**Purpose:** Lists all non-router skills with line-count + leak-count. Used by the pre-commit hook (Plan 07) as a warning (non-blocking) indicator of skill bloat.

**Implementation:**

```typescript
import { execSync } from 'node:child_process';
import { readdirSync, readFileSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { GSDError, ErrorClassification } from '../errors.js';
import type { QueryHandler } from './utils.js';

interface SkillReport {
  name: string;
  path: string;
  lines: number;
  leaks: number;
  isRouter: boolean;
}

export const verifyFatSkills: QueryHandler = async (args, projectDir) => {
  // 1. Discover all skill files (workflows + commands + agents)
  //    Look in the GSD install directory for .md files
  const gsdDir = join(projectDir, 'get-shit-done');
  const skillDirs = ['workflows', 'commands', 'agents'];
  
  const skills: SkillReport[] = [];
  
  for (const dir of skillDirs) {
    const dirPath = join(gsdDir, dir);
    let files: string[];
    try {
      files = readdirSync(dirPath).filter(f => f.endsWith('.md'));
    } catch { continue; }
    
    for (const file of files) {
      const filePath = join(dirPath, file);
      const content = readFileSync(filePath, 'utf-8');
      const lines = content.split('\n').length;
      
      // Detect router skills (those that just dispatch to other skills)
      const isRouter = content.includes('route-next-action') || lines < 20;
      
      // Count leaks via leak-grep
      let leaks = 0;
      try {
        const result = execSync(`node scripts/leak-grep.cjs "${filePath}" 2>/dev/null`, {
          cwd: projectDir,
          encoding: 'utf-8',
          timeout: 5000,
        });
        leaks = result.trim().split('\n').filter(l => l.length > 0).length;
      } catch (e: any) {
        // leak-grep exits 1 when leaks found — parse stdout
        if (e.stdout) {
          leaks = e.stdout.trim().split('\n').filter((l: string) => l.length > 0).length;
        }
      }
      
      skills.push({ name: file.replace('.md', ''), path: `${dir}/${file}`, lines, leaks, isRouter });
    }
  }
  
  // 2. Filter to non-router skills and sort by line count descending
  const nonRouter = skills.filter(s => !s.isRouter).sort((a, b) => b.lines - a.lines);
  
  // 3. Format output
  const thresholdLines = 500; // warn above this
  const warnings = nonRouter.filter(s => s.lines > thresholdLines || s.leaks > 0);
  
  return {
    data: {
      total_skills: skills.length,
      non_router_skills: nonRouter.length,
      over_threshold: warnings.length,
      threshold_lines: thresholdLines,
      skills: nonRouter.map(s => ({
        name: s.name,
        path: s.path,
        lines: s.lines,
        leaks: s.leaks,
        warning: s.lines > thresholdLines ? 'fat' : s.leaks > 0 ? 'leaky' : null,
      })),
    },
  };
};
```

NOTE: This handler reads skill files from the GSD install directory using raw fs — this is intentional and NOT a leak. It reads `get-shit-done/workflows/*.md` (the skill source files), NOT `.planning/` files. The REQUIREMENTS.md Out of Scope section confirms: "detectCustomFiles, agentSkills, skillManifest (read side) (runtime install dirs and skill roots)" are C2.
