```typescript
export const phaseAdd: QueryHandler = async (args, projectDir, workstream) => {
  const description = args[0];
  if (!description) throw new GSDError('description required', ErrorClassification.Validation);
  assertNoNullBytes(description, 'description');
  
  const adapter = await adapterFor(projectDir);
  const slug = generateSlugInternal(description);
  const nextNum = await scanNextPhaseNumber(adapter, workstream);
  const dirName = `${String(nextNum).padStart(2, '0')}-${slug}`;
  assertSafePhaseDirName(dirName);
  
  await adapter.withTransaction(async () => {
    await scaffoldPhaseDir(adapter, workstream, dirName);
    const phaseEntry = buildPhaseEntry(nextNum, description, dirName);
    await insertRoadmapPhase(adapter, workstream, phaseEntry);
    await updateStateProgressFields(adapter, workstream, { totalPhases: nextNum }, projectDir);
  });
  
  return { data: { phase: nextNum, slug, dirName } };
};
```
