    /** Transaction context: outer-most call allocates snapshot; nested calls no-op (reentrancy). */
    interface TxnContext {
      depth: number;
      snapshotPath: string | null;
    }

    const txnStacks = new WeakMap<BdRunner, TxnContext>();

    export async function snapshot(bd: BdRunner): Promise<string> {
      const snapPath = join(tmpdir(), `bd-snap-${process.pid}-${randomBytes(6).toString('hex')}.jsonl`);
      bd.run(['export', '--json', '-o', snapPath], { parseJson: false });
      return snapPath;
    }

    function derivePrefixFromSnapshot(snapshotPath: string): string {
      // WR-04 fix: read first issue line, extract prefix from id (e.g., "sd-001" → "sd", "alpha-42" → "alpha")
      if (!existsSync(snapshotPath)) return 'bd';  // safe fallback
      const content = readFileSync(snapshotPath, 'utf-8');
      const firstLine = content.split('\n').find((l) => l.trim().length > 0 && l.trim().startsWith('{'));
      if (!firstLine) return 'bd';
      try {
        const record = JSON.parse(firstLine) as { id?: string };
        const m = record.id ? /^([a-z]+)-\d+$/.exec(record.id) : null;
        return m ? m[1]! : 'bd';
      } catch {
        return 'bd';
      }
    }

    export async function restore(state: BeadsRuntimeState, snapshotId: string): Promise<void> {
      const prefix = derivePrefixFromSnapshot(snapshotId);
      const tmpDir = mkdtempSync(join(tmpdir(), 'bd-restore-'));
      try {
        // bd init --from-jsonl into fresh tmp dir (creates tmpDir/.beads/)
        state.bd.run([

          '--non-interactive', '--skip-agents', '--skip-hooks', '--quiet',
        ], { parseJson: false, env: { BEADS_ACTOR: 'seed' } });

        // Actually, bd init runs in the tmpDir cwd — we need a dedicated BdRunner for it
        // [EXECUTOR: adjust by spawning bd directly with cwd=tmpDir, NOT using state.bd which has project's cwd baked]
        //
        // Simpler: use child_process.spawnSync directly here so we control cwd
        // [full impl per sibling primitives.mjs:435-494 pattern]

        // POSIX atomic cutover: rename project's .beads/ out, rename tmpDir/.beads/ in
        const projectBeads = state.beadsRoot;
        const parkedPath = join(tmpDir, 'old-beads');
        renameSync(projectBeads, parkedPath);
        renameSync(join(tmpDir, '.beads'), projectBeads);
      } finally {
        rmSync(tmpDir, { recursive: true, force: true });
      }

    export async function withTransaction<T>(state: BeadsRuntimeState, fn: () => Promise<T>): Promise<T> {
      let ctx = txnStacks.get(state.bd);
      if (!ctx) {
        ctx = { depth: 0, snapshotPath: null };
        txnStacks.set(state.bd, ctx);
      }

      // Reentrancy: nested calls join the outer transaction
      if (ctx.depth > 0) {
        ctx.depth++;
        try {
          return await fn();
        } finally {
          ctx.depth--;
        }
      }

      // Outer-most call: allocate snapshot
      ctx.depth = 1;
      ctx.snapshotPath = await snapshot(state.bd);
      try {
        const result = await fn();
        // Commit = discard snapshot
        rmSync(ctx.snapshotPath, { force: true });
        ctx.snapshotPath = null;
        return result;
      } catch (e) {
        // Rollback — restore from snapshot
        if (ctx.snapshotPath) {
          await restore(state, ctx.snapshotPath);
          rmSync(ctx.snapshotPath, { force: true });
          ctx.snapshotPath = null;
        }
        throw e;
      } finally {
        ctx.depth--;
      }
    }

    async function _resolveMilestoneBead(state: BeadsRuntimeState, milestoneKey: string): Promise<string | null> {
      try {
        const list = state.bd.run(['list', '-l', 'gsd:milestone', '-l', `version:${milestoneKey}`, '--json', '--all', '-n', '0']) as Array<{ id: string }>;

    function _deriveEventId(payload: Record<string, unknown>): string {
      // Shorten a stable ID from payload content for dedupe key scoping
      const json = JSON.stringify(payload);
      let h = 0;
      for (const ch of json) h = ((h << 5) - h + ch.charCodeAt(0)) | 0;
      return Math.abs(h).toString(36);

    export async function recordStateAppend(state: BeadsRuntimeState, event: AppendEvent): Promise<StateWriteOutcome> {
      return withTransaction(state, async () => {
        switch (event.type) {
          case 'session':
          case 'quick_task':
          case 'forensic_session': {
            // High-frequency → bd comments add --author gsd:event:<type>
            const milestone = await _getCurrentMilestone(state);
            const milestoneBead = await _resolveMilestoneBead(state, milestone);
            if (!milestoneBead) {
              // No milestone bead yet — create one? Or fail? For v1.0 MVP: fail soft
              return { applied: false, reason: 'nothing_to_remove' };  // somewhat misuse, but safer than silent success
            }
            // Dedupe: list existing comments + author-match
            let existing: Array<{ author: string; text: string }> = [];
            try {
              existing = state.bd.run(['comments', milestoneBead, '--json']) as typeof existing;
            } catch (e) {
              if (!(e instanceof BeadsEmpty)) throw e;
            }
            const author = `gsd:event:${event.type}`;
            const payloadJson = JSON.stringify(event.payload);
            const dupe = existing.find((c) => c.author === author && c.text === payloadJson);
            if (dupe) return { applied: false, reason: 'duplicate' };

            state.bd.run(['comments', 'add', milestoneBead, '--author', author, payloadJson], { parseJson: false });
            return { applied: true };
          }

          case 'decision':
          case 'metric':
          case 'roadmap_evolution': {
            // Low-frequency → bd remember <json> --key <milestone>:<type>:<id>
            const milestone = await _getCurrentMilestone(state);
            const eventId = _deriveEventId(event.payload as Record<string, unknown>);
            const key = `${milestone}:${event.type}:${eventId}`;

            const recall = state.bd.run(['recall', key], { parseJson: false });
            if (recall !== null && typeof recall === 'string' && recall.trim().length > 0) {
              return { applied: false, reason: 'duplicate' };
            }
            state.bd.run(['remember', JSON.stringify(event.payload), '--key', key], { parseJson: false });
            return { applied: true };
          }

          default: {
            const _exhaustive: never = event;
            throw new Error(`Unknown AppendEvent: ${(event as { type: string }).type}`);
          }
        }
      });

    export async function recordStateMutation(state: BeadsRuntimeState, event: MutationEvent): Promise<StateWriteOutcome> {
      return withTransaction(state, async () => {
        switch (event.type) {
          case 'blocker_added': {

            const milestoneBead = await _resolveMilestoneBead(state, milestone);
            if (!milestoneBead) return { applied: false, reason: 'nothing_to_remove' };
            const label = `gsd:blocker:${_deriveEventId({ text: event.payload.text })}`;
            // Check existing labels via bd show
            const shown = state.bd.show(milestoneBead);
            const existingLabels = (shown.labels as string[] | undefined) ?? [];
            if (existingLabels.includes(label)) return { applied: false, reason: 'duplicate' };
            state.bd.run(['label', 'add', milestoneBead, label], { parseJson: false });
            return { applied: true };
          }
          case 'blocker_resolved': {
            const milestone = await _getCurrentMilestone(state);
            const milestoneBead = await _resolveMilestoneBead(state, milestone);
            if (!milestoneBead) return { applied: false, reason: 'nothing_to_remove' };
            const label = `gsd:blocker:${_deriveEventId({ text: event.payload.text })}`;
            const shown = state.bd.show(milestoneBead);
            const existingLabels = (shown.labels as string[] | undefined) ?? [];
            if (!existingLabels.includes(label)) return { applied: false, reason: 'nothing_to_remove' };
            state.bd.run(['label', 'remove', milestoneBead, label], { parseJson: false });
            return { applied: true };
          }
          case 'todo_count_update': {
            const milestone = await _getCurrentMilestone(state);
            const key = `${milestone}:todo_count:current`;
            const recall = state.bd.run(['recall', key], { parseJson: false });
            const prev = recall !== null && typeof recall === 'string' ? recall.trim() : null;
            const next = JSON.stringify(event.payload);
            if (prev === next) return { applied: false, reason: 'duplicate' };
            state.bd.run(['remember', next, '--key', key], { parseJson: false });
            return { applied: true };
          }
          case 'deferred_items': {
            const milestone = await _getCurrentMilestone(state);
            const key = `${milestone}:deferred_items:${event.payload.action}`;
            const next = JSON.stringify(event.payload.items);
            const recall = state.bd.run(['recall', key], { parseJson: false });
            const prev = recall !== null && typeof recall === 'string' ? recall.trim() : null;
            if (event.payload.action === 'add') {
              if (prev === next) return { applied: false, reason: 'duplicate' };
              state.bd.run(['remember', next, '--key', key], { parseJson: false });
              return { applied: true };
            } else {
              // remove
              if (prev === null) return { applied: false, reason: 'nothing_to_remove' };
              state.bd.run(['forget', key], { parseJson: false });
              return { applied: true };
            }
          }
          default: {
            const _: never = event;
            throw new Error(`Unknown MutationEvent: ${(event as { type: string }).type}`);
          }
        }
      });
    }

    export async function recordStateSignal(state: BeadsRuntimeState, event: SignalEvent): Promise<StateWriteOutcome> {
      return withTransaction(state, async () => {
        switch (event.type) {
          case 'waiting': {
            const milestone = await _getCurrentMilestone(state);
            const milestoneBead = await _resolveMilestoneBead(state, milestone);
            if (!milestoneBead) return { applied: false, reason: 'nothing_to_remove' };
            const label = `gsd:waiting:${event.payload.waitType}`;
            const shown = state.bd.show(milestoneBead);
            const existingLabels = (shown.labels as string[] | undefined) ?? [];
            if (existingLabels.includes(label)) return { applied: false, reason: 'duplicate' };
            state.bd.run(['label', 'add', milestoneBead, label], { parseJson: false });
            return { applied: true };
          }
          case 'resume': {
            const milestone = await _getCurrentMilestone(state);
            const milestoneBead = await _resolveMilestoneBead(state, milestone);
            if (!milestoneBead) return { applied: false, reason: 'nothing_to_remove' };
            // Find + remove any gsd:waiting:* labels
            const shown = state.bd.show(milestoneBead);
            const existingLabels = (shown.labels as string[] | undefined) ?? [];
            const waiting = existingLabels.filter((l) => l.startsWith('gsd:waiting:'));
            if (waiting.length === 0) return { applied: false, reason: 'nothing_to_remove' };
            for (const label of waiting) {
              state.bd.run(['label', 'remove', milestoneBead, label], { parseJson: false });
            }
            return { applied: true };
          }
          default: {
            const _: never = event;
            throw new Error(`Unknown SignalEvent: ${(event as { type: string }).type}`);
          }
        }
      });
    }

    /**
     * Dependency-edge synthesizer — BEADS-03 / D-OQ06.
     *
     * Produces {type: 'dependency', confidence: 1.0} edges from bd's blocks edges.
     *
     * Intercepts getRecord('graphs/graph.json') at the primitives layer and
     * materializes the edge list lazily. Claude's discretion per RESEARCH Pattern 4:
     * "lazy-with-memoization-within-txn" is the recommended strategy; current impl
     * is simply lazy (one bd export per call).
     *
     * Spike-014 semantics (carried forward):
     *   - Field name in bd export is `type`, NOT `dependency_type` (which is a bd show
     *     shape quirk).
     *   - Direction: `depends_on_id` is the blocker (from side).
     *   - Single `bd export --json` surfaces all edges (fits ≤2-spawn budget).
     *   - Cascade-loop walks parent-child only — blocks edges don't trigger cascade.
     */
    import type { BdRunner } from './bd/helper.js';
    import { BeadsEmpty } from './bd/errors.js';

    export interface DepGraphEdge {
      from: string;
      to: string;
      type: 'dependency';
      confidence: 1.0;
    }

    export interface DepGraphPayload {
      edges: DepGraphEdge[];
      // Additional fields the fork's graph.json might expect (nodes, metadata) — leave open
    }

    export async function materializeGraphJson(bd: BdRunner): Promise<string> {
      let beads: Array<{
        _type: string;
        id: string;
        dependencies?: Array<{ depends_on_id: string; type: string }>;
      }> = [];
      try {
        const result = bd.run(['export', '--json']);
        beads = Array.isArray(result) ? result as typeof beads : [result] as typeof beads;
      } catch (e) {
        if (e instanceof BeadsEmpty) {
          // Empty store: emit valid JSON with no edges
          return JSON.stringify({ edges: [] } satisfies DepGraphPayload);
        }
        throw e;
      }

      const issues = beads.filter((b) => b._type === 'issue');
      const depEdges: DepGraphEdge[] = issues.flatMap((issue) =>
        (issue.dependencies ?? [])
          .filter((d) => d.type === 'blocks')               // spike-014: 'blocks' not 'dependency_type'
          .map((d) => ({
            from: d.depends_on_id,                           // blocker direction per sibling DP-02
            to: issue.id,
            type: 'dependency' as const,
            confidence: 1.0,
          }))
      );

      return JSON.stringify({ edges: depEdges } satisfies DepGraphPayload);
    }
