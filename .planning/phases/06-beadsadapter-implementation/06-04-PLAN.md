     * Path router — classifies .planning/ relative paths into routing records.
     *
     * Ported from sibling src/adapter/pathRouter.mjs (97 LOC). Amendments:
     *

     *      optional bd-memory index key as a post-write side-effect; no

     *   4. Canonical-file list extended to fork's shape (REQUIREMENTS.md,
     *      DECISIONS.md, PROJECT.md, STATE.md, HANDOFF.json, CONTINUE-HERE.json).

    import type { NamedDocCategory, RootNamedDocKey } from 'get-shit-done/adapters/types.js';

    export type Tier = 'bd' | 'disk';

    export interface Route {
      kind: RouteKind;
      tier: Tier;
      label?: string;
      singleton?: boolean;
      collection?: boolean;
      phase?: string;
      plan?: string;
      category?: NamedDocCategory;
      key?: string | RootNamedDocKey;

      | 'roadmap-singleton'
      | 'state-singleton'
      | 'project-singleton'
      | 'requirements-singleton'
      | 'decisions-singleton'
      | 'phase-context'
      | 'phase-research'
      | 'phase-discussion'
      | 'phase-plan'
      | 'phase-summary'
      | 'phase-uat'
      | 'phase-verification'
      | 'phase-collection'
      | 'named-doc'
      | 'named-doc-root'
      | 'opaque';

      'research', 'intel', 'codebase', 'archived-milestone', 'reports', 'sketches', 'tmp', 'root',
    ]);

    /**
     * Classify a .planning/-relative path into a Route.
     * Input paths are normalized (no leading ./ or ../).
     */
    export function resolveRoute(path: string): Route {
      // Normalize: strip leading './', reject absolute/parent-escape (defense-in-depth;
      // primitive-level _abs() guard in Plan 06-05 is the authoritative check).
      const p = path.replace(/^\.\//, '');

      // 1. Top-level singletons — bd-tier (label-dispatched via bd list -l gsd:<name>)
      if (p === 'ROADMAP.md') return { kind: 'roadmap-singleton', tier: 'bd', label: 'gsd:roadmap', singleton: true };
      if (p === 'STATE.md') return { kind: 'state-singleton', tier: 'bd', label: 'gsd:state', singleton: true };
      if (p === 'PROJECT.md') return { kind: 'project-singleton', tier: 'bd', label: 'gsd:project', singleton: true };
      if (p === 'REQUIREMENTS.md') return { kind: 'requirements-singleton', tier: 'bd', label: 'gsd:requirements', singleton: true };
      if (p === 'DECISIONS.md') return { kind: 'decisions-singleton', tier: 'bd', label: 'gsd:decisions', singleton: true };

      // 2. Phase-addressed paths: phases/<NN-slug>/...
      const phaseMatch = /^phases\/([^/]+)(?:\/(.+))?$/.exec(p);
      if (phaseMatch) {
        const phase = phaseMatch[1]!;
        const rest = phaseMatch[2];
        if (!rest) return { kind: 'phase-collection', tier: 'bd', label: 'gsd:phase', phase, collection: true };

        // Phase documents: <phase>-CONTEXT.md, <phase>-RESEARCH.md, <phase>-DISCUSSION-LOG.md, <phase>-NN-PLAN.md, <phase>-NN-SUMMARY.md, <phase>-UAT.md, <phase>-VERIFICATION.md
        if (/-CONTEXT\.md$/.test(rest)) return { kind: 'phase-context', tier: 'bd', phase, singleton: true };
        if (/-RESEARCH\.md$/.test(rest)) return { kind: 'phase-research', tier: 'bd', phase, singleton: true };
        if (/-DISCUSSION-LOG\.md$/.test(rest)) return { kind: 'phase-discussion', tier: 'bd', phase, singleton: true };
        const planMatch = /-(\d+)-PLAN\.md$/.exec(rest);
        if (planMatch) return { kind: 'phase-plan', tier: 'bd', phase, plan: planMatch[1]! };
        const summaryMatch = /-(\d+)-SUMMARY\.md$/.exec(rest);
        if (summaryMatch) return { kind: 'phase-summary', tier: 'bd', phase, plan: summaryMatch[1]! };
        if (/-UAT\.md$/.test(rest)) return { kind: 'phase-uat', tier: 'bd', phase, singleton: true };
        if (/-VERIFICATION\.md$/.test(rest)) return { kind: 'phase-verification', tier: 'bd', phase, singleton: true };

        // Phase-scoped named docs (sketches, scratch, etc.) — disk-tier per CR-02 resolution
        return { kind: 'opaque', tier: 'disk' };
      }

      // 3. Named-doc categories (disk-tier per CR-02 resolution)
      for (const category of NAMED_DOC_CATEGORIES) {
        if (category === 'root') continue;  // handled next
        const rx = new RegExp(`^${category}\\/(.+)$`);
        const m = rx.exec(p);
        if (m) return { kind: 'named-doc', tier: 'disk', category, key: m[1]! };
      }

      // 4. Root-category singletons (HANDOFF / CONTINUE-HERE / DECISIONS-INDEX)
      if (p === 'HANDOFF.json' || p === 'HANDOFF.md') return { kind: 'named-doc-root', tier: 'disk', category: 'root', key: 'HANDOFF' };
      if (p === 'CONTINUE-HERE.json' || p === 'CONTINUE-HERE.md') return { kind: 'named-doc-root', tier: 'disk', category: 'root', key: 'CONTINUE-HERE' };
      if (p === 'DECISIONS-INDEX.md' || p === 'DECISIONS-INDEX.json') return { kind: 'named-doc-root', tier: 'disk', category: 'root', key: 'DECISIONS-INDEX' };

      // 5. Fall-through
      return { kind: 'opaque', tier: 'disk' };
    }
