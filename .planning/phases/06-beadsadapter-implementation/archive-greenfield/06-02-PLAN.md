**(1) `src/primitives/record.ts` — the 7 primitives.** All methods take a `BdClient` instance (passed in from BeadsAdapter). Implement as free functions so the module is pure + composable:

```typescript
import type { RecordRef, RecordFilter } from 'get-shit-done/adapters/types.js';
import type { BdClient } from '../bd-client.js';

const PATH_LABEL_PREFIX = 'gsd:path:';

function pathLabel(path: string): string {
  return `${PATH_LABEL_PREFIX}${path}`;
}

export async function getRecord(bdClient: BdClient, path: string): Promise<string | null> {
  const issue = await bdClient.issueFindByLabel(pathLabel(path));
  if (issue === null) return null;
  return issue.body;
}

export async function putRecord(bdClient: BdClient, path: string, body: string): Promise<void> {
  const existing = await bdClient.issueFindByLabel(pathLabel(path));
  if (existing !== null) {
    await bdClient.issueUpdate(existing.id, body);
  } else {
    // Title convention: last path segment. Labels: [gsd:path:<path>].
    const title = path.split('/').pop() ?? path;
    await bdClient.issueCreate(title, body, [pathLabel(path)]);
  }
}

export async function removeRecord(bdClient: BdClient, path: string): Promise<void> {
  const existing = await bdClient.issueFindByLabel(pathLabel(path));
  if (existing === null) return;  // null-on-miss matches MarkdownAdapter
  await bdClient.issueDelete(existing.id);
}

export async function removeCollection(bdClient: BdClient, prefix: string): Promise<void> {
  // Find every issue whose path starts with `prefix/` (or equals `prefix`).
  // Use a label-prefix search: `gsd:path:<prefix>/` matches children.
  const normalizedPrefix = prefix.endsWith('/') ? prefix : `${prefix}/`;
  const items = await bdClient.issueListByLabelPrefix(`${PATH_LABEL_PREFIX}${normalizedPrefix}`);
  // Also include an exact-match of the prefix itself (in case prefix is also a record path).
  const exact = await bdClient.issueFindByLabel(pathLabel(prefix));
  if (exact !== null) items.push({ id: exact.id, labels: [], mtime: exact.mtime });
  for (const item of items) {
    await bdClient.issueDelete(item.id);
  }
}

export async function listCollection(
  bdClient: BdClient,
  prefix: string,
  filter?: RecordFilter,
): Promise<RecordRef[]> {
  // Return immediate children under `prefix` (non-recursive). Matches
  // MarkdownAdapter readdir semantics: one level only.
  const normalizedPrefix = prefix.endsWith('/') ? prefix : `${prefix}/`;
  const items = await bdClient.issueListByLabelPrefix(`${PATH_LABEL_PREFIX}${normalizedPrefix}`);

  const seenImmediateChildren = new Set<string>();
  const refs: RecordRef[] = [];
  for (const item of items) {
    const pathLabelEntry = item.labels.find(l => l.startsWith(PATH_LABEL_PREFIX));
    if (!pathLabelEntry) continue;
    const fullPath = pathLabelEntry.slice(PATH_LABEL_PREFIX.length);
    // strip prefix, take first segment — emulates fs.readdir "immediate children"
    const remainder = fullPath.slice(normalizedPrefix.length);
    if (!remainder || remainder.includes('/') === false) {
      // it's a direct-child FILE
      if (!seenImmediateChildren.has(remainder)) {
        seenImmediateChildren.add(remainder);
        refs.push({ path: fullPath, name: remainder });
      }
    } else {
      // deeper nesting — emit the intermediate "dir" child once
      const firstSegment = remainder.split('/')[0];
      if (!seenImmediateChildren.has(firstSegment)) {
        seenImmediateChildren.add(firstSegment);
        refs.push({
          path: `${normalizedPrefix}${firstSegment}`,
          name: firstSegment,
        });
      }
    }
  }
  return filter ? refs.filter(filter) : refs;
}

export async function exists(bdClient: BdClient, path: string): Promise<boolean> {
  const issue = await bdClient.issueFindByLabel(pathLabel(path));
  return issue !== null;
}

export async function stat(
  bdClient: BdClient,
  path: string,
): Promise<{ kind: 'file' | 'dir'; mtime?: string } | null> {
  // Try exact-match first → it's a file.
  const exact = await bdClient.issueFindByLabel(pathLabel(path));
  if (exact !== null) {
    return { kind: 'file', mtime: exact.mtime };
  }
  // No exact match → check if any issue exists under `path/` → it's a "dir".
  const normalizedPrefix = path.endsWith('/') ? path : `${path}/`;
  const items = await bdClient.issueListByLabelPrefix(`${PATH_LABEL_PREFIX}${normalizedPrefix}`);
  if (items.length > 0) {
    // Synthetic dir — no mtime (bd has no directory concept; omit per contract).
    return { kind: 'dir' };
  }
  return null;
}
```
