```typescript
export class UnknownSectionError extends Error {
  override readonly name = 'UnknownSectionError';
  readonly code = 'UNKNOWN_SECTION' as const;
  readonly path: string;
  readonly heading: string;
  readonly expectedSchema: string;
  readonly __brand = 'UnknownSectionError' as const;

  constructor(path: string, heading: string, expectedSchema: string) {
    super(
      `${path} contains L2 heading "${heading}" not declared in schema ${expectedSchema}. ` +
      `Add the heading to ${expectedSchema} or rename it to a known heading.`
    );
    this.path = path;
    this.heading = heading;
    this.expectedSchema = expectedSchema;
  }

  static [Symbol.hasInstance](instance: unknown): boolean {
    return (
      instance != null &&
      typeof instance === 'object' &&
      (instance as Record<string, unknown>).__brand === 'UnknownSectionError'
    );
  }
}
```

```typescript
import type { FormatSchema } from './types.js';  // or inline if ergonomic

export interface FormatSchema<S = unknown> {
  matches(path: string): boolean;
  parse(path: string, markdown: string): S;
  render(schema: S, frontmatter: Record<string, unknown>): string;
}

// Populated by individual schema modules' side-effect imports, OR imported
// explicitly and array-assembled here. Prefer explicit assembly:
import { stateSchema } from './state.js';  // stub until Task 2
// ... etc; Task 2 uncomments as each schema lands

export const formatRegistry: FormatSchema[] = [
  // stateSchema, roadmapSchema, ...
];

export function findSchema(path: string): FormatSchema | null {
  for (const s of formatRegistry) {
    if (s.matches(path)) return s;
  }
  return null;
}

// Fallback "passthrough" schema for paths that don't match any canonical
// file. Used so BeadsAdapter can handle arbitrary .planning paths (e.g.
// HANDOFF.json, tmp/*, unknown scratch docs) without failing. Parse =
// wrap body in { raw: body }, render = return raw.
export const passthroughSchema: FormatSchema<{ raw: string }> = {
  matches: () => true,
  parse: (_path, md) => ({ raw: md }),
  render: (s) => s.raw,
};

export function findSchemaOrPassthrough(path: string): FormatSchema {
  return findSchema(path) ?? passthroughSchema;
}
```

```typescript
import type { SectionMode } from 'get-shit-done/adapters/types.js';
import type { BdClient } from '../bd-client.js';
import { getRecord, putRecord } from './record.js';
import { extractSection, replaceSection } from '../format/heading-walker.js';

export async function getSection(bdClient: BdClient, path: string, anchor: string): Promise<string | null> {
  const text = await getRecord(bdClient, path);
  if (text === null) return null;
  const { body } = extractSection(text, anchor);
  return body;
}

export async function updateSection(
  bdClient: BdClient,
  path: string,
  anchor: string,
  body: string,
  mode: SectionMode,
): Promise<void> {
  // NOTE: Plan 04 wraps this in withTransaction. Until then, the operation is
  // non-atomic w.r.t. other writers. Smoke tests run single-writer so this
  // does not yet cause visible issues; conformance's transaction tests in
  // Plan 04 close the gap.
  const current = (await getRecord(bdClient, path)) ?? '';
  const { body: existingBody, found } = extractSection(current, anchor);

  let next: string;
  if (!found) {
    const sep = current.endsWith('\n') ? '' : '\n';
    next = `${current}${sep}\n${anchor}\n\n${body}\n`;
  } else {
    let newBody: string;
    switch (mode) {
      case 'overwrite':
        newBody = body;
        break;
      case 'append':
        newBody = existingBody !== null && existingBody.length > 0
          ? `${existingBody}\n\n${body}`
          : body;
        break;
      case 'prepend':
        newBody = existingBody !== null && existingBody.length > 0
          ? `${body}\n\n${existingBody}`
          : body;
        break;
      default: {
        const _exhaustive: never = mode;
        throw new Error(`Unknown SectionMode: ${String(_exhaustive)}`);
      }
    }
    next = replaceSection(current, anchor, newBody);
  }
  await putRecord(bdClient, path, next);
}
```

```typescript
import type { BdClient } from '../bd-client.js';
import { getRecord, putRecord } from './record.js';
import { parseFrontmatter, stringifyFrontmatter } from '../format/frontmatter.js';

export async function getFrontmatter(
  bdClient: BdClient,
  path: string,
  field?: string,
): Promise<unknown> {
  const text = await getRecord(bdClient, path);
  if (text === null) return field !== undefined ? null : {};
  const { frontmatter } = parseFrontmatter(text);
  return field !== undefined ? (frontmatter[field] ?? null) : frontmatter;
}

export async function updateFrontmatter(
  bdClient: BdClient,
  path: string,
  field: string,
  value: unknown,
): Promise<void> {
  const text = (await getRecord(bdClient, path)) ?? '';
  const { frontmatter, body } = parseFrontmatter(text);
  frontmatter[field] = value;
  await putRecord(bdClient, path, stringifyFrontmatter(frontmatter) + body);
}

export async function mergeFrontmatter(
  bdClient: BdClient,
  path: string,
  patch: Record<string, unknown>,
): Promise<void> {
  const text = (await getRecord(bdClient, path)) ?? '';
  const { frontmatter, body } = parseFrontmatter(text);
  const merged = { ...frontmatter, ...patch };
  await putRecord(bdClient, path, stringifyFrontmatter(merged) + body);
}
```
