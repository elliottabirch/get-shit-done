**(5) Implement `src/foundational/transaction.ts`:**
```typescript
import type { TxnCtx } from '../txn/index.js';
import * as txn from '../txn/index.js';

export interface WithTransactionOpts {
  dryRun?: boolean;
}

// Internal type — only imported by BeadsAdapter + tests
export interface TransactionalAdapter {
  readonly bdClient: { /* ... */ };
  activeTxn: TxnCtx | undefined;
}

export async function withTransaction<T>(
  adapter: TransactionalAdapter,
  fn: () => Promise<T>,
  opts?: WithTransactionOpts,
): Promise<T> {
  // D-10 reentrant guard
  if (adapter.activeTxn) {
    adapter.activeTxn.depth += 1;
    try {
      return await fn();
    } finally {
      adapter.activeTxn!.depth -= 1;
    }
  }
  // Outer txn
  const ctx = txn.beginTransaction(/* projectDir or bdClient as needed */);
  ctx.dryRun = opts?.dryRun ?? false;
  ctx.depth = 1;
  adapter.activeTxn = ctx;
  let succeeded = false;
  try {
    const result = await fn();
    succeeded = true;
    if (!ctx.dryRun) {
      await txn.commit(adapter.bdClient, ctx);
    } else {
      await txn.rollback(ctx); // dryRun discards on exit per D-05
    }
    return result;
  } catch (err) {
    await txn.rollback(ctx);
    throw err;
  } finally {
    adapter.activeTxn = undefined;
  }
}
```
