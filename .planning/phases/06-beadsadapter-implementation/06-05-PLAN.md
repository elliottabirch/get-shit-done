export interface BeadsRuntimeState {
  beadsRoot: string;
  bd: BdRunner;
}

export async function ensureBd(projectRoot: string, cached: BeadsRuntimeState | null): Promise<BeadsRuntimeState> {
  if (cached) return cached;
  const root = findBeadsRoot(projectRoot);
  if (!root) {

      projectRoot,
      'Expected .beads/metadata.json at or above projectRoot. Run `bd init` or set BEADS_DIR.',

  return { beadsRoot: root, bd: new BdRunner(root) };
}

      if (!existsSync(abs)) return null;
      return readFileSync(abs, 'utf-8');
    }

    function _namedDocPath(category: NamedDocCategory, key: string, workstream?: string): string {
      if (category === 'root') {
        // RootNamedDocKey — HANDOFF / CONTINUE-HERE / DECISIONS-INDEX
        const ext = key === 'HANDOFF' || key === 'CONTINUE-HERE' ? '.json' : '.md';
        return `${key}${ext}`;
      }
      const ws = workstream ? `${workstream}/` : '';
      return `${category}/${ws}${key}`;
    }
