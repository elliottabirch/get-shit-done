import type { AppendEvent, MutationEvent, SignalEvent } from 'get-shit-done/adapters/state-event-types.js';
import type { StateWriteOutcome } from 'get-shit-done/adapters/types.js';
import type { TransactionalAdapter } from '../foundational/transaction.js';

import { getRecord, putRecord, removeRecord, exists } from './record.js';
import { getFrontmatter } from './frontmatter.js';
import * as stateFormat from '../format/state.js';

export async function recordStateAppend(
  adapter: TransactionalAdapter,
  event: AppendEvent,
): Promise<StateWriteOutcome> {
  return withTransaction(adapter, async () => {
    const raw = (await getRecord(adapter, 'STATE.md')) ?? '';
    const { body } = /* parseFrontmatter split */;
    const schema = stateFormat.parseStateMd('STATE.md', raw);
    const fm = (await getFrontmatter(adapter, 'STATE.md')) as Record<string, unknown>;

    switch (event.type) {
      case 'decision': {
        const { phase, summary, rationale } = event.payload;
        if (stateFormat.isDecisionDuplicate(schema, phase, summary)) {
          return { applied: false, reason: 'duplicate' };
        }
        const sectionAbsent = schema.decisions.length === 0 && !raw.includes('## Decisions Made');
        schema.decisions.push({ phase, summary, rationale });
        const rendered = stateFormat.renderStateMd(schema, fm);
        await putRecord(adapter, 'STATE.md', rendered);
        return sectionAbsent
          ? { applied: true, created_section: '## Decisions Made' }
          : { applied: true };
      }

      case 'metric': {
        const sectionAbsent = !raw.includes('## Performance Metrics');
        schema.metrics.push(event.payload);
        const rendered = stateFormat.renderStateMd(schema, fm);
        await putRecord(adapter, 'STATE.md', rendered);
        return sectionAbsent
          ? { applied: true, created_section: '## Performance Metrics' }
          : { applied: true };
      }

      case 'roadmap_evolution': {
        // Dedupe scope: `### Roadmap Evolution` subsection only (WR-02).
        const entry = event.payload;
        const rendered = stateFormat.renderRoadmapEvolutionBullet(entry);
        if (stateFormat.isRoadmapEvolutionDuplicate(schema, rendered)) {
          return { applied: false, reason: 'duplicate' };
        }
        const subsectionAbsent = !raw.includes('### Roadmap Evolution');
        schema.accumulatedContext = schema.accumulatedContext ?? {};
        schema.accumulatedContext.roadmapEvolution = [
          ...(schema.accumulatedContext.roadmapEvolution ?? []),
          entry,
        ];
        const out = stateFormat.renderStateMd(schema, fm);
        await putRecord(adapter, 'STATE.md', out);
        return subsectionAbsent
          ? { applied: true, created_section: '### Roadmap Evolution' }
          : { applied: true };
      }

      case 'session': {
        const sectionAbsent = !raw.includes('## Session Continuity');
        schema.sessionContinuity = { ...(schema.sessionContinuity ?? {}), ...event.payload };
        const rendered = stateFormat.renderStateMd(schema, fm);
        await putRecord(adapter, 'STATE.md', rendered);
        return sectionAbsent
          ? { applied: true, created_section: '## Session Continuity' }
          : { applied: true };
      }

      case 'forensic_session': {
        schema.forensicSessions = [...(schema.forensicSessions ?? []), event.payload];
        await putRecord(adapter, 'STATE.md', stateFormat.renderStateMd(schema, fm));
        return { applied: true };
      }

      case 'quick_task': {
        schema.quickTasks = [...(schema.quickTasks ?? []), event.payload];
        await putRecord(adapter, 'STATE.md', stateFormat.renderStateMd(schema, fm));
        return { applied: true };
      }

      default: {
        const _exhaustive: never = event;
        throw new Error(`Unknown AppendEvent type: ${(event as { type: string }).type}`);
      }
    }
  });
}

```typescript
export async function recordStateSignal(
  adapter: TransactionalAdapter,
  event: SignalEvent,
): Promise<StateWriteOutcome> {
  return withTransaction(adapter, async () => {
    switch (event.type) {
      case 'waiting': {
        await putRecord(adapter, 'WAITING.json', JSON.stringify({ status: 'waiting', ...event.payload }));
        return { applied: true };
      }
      case 'resume': {
        if (!(await exists(adapter, 'WAITING.json'))) {
          return { applied: false, reason: 'nothing_to_remove' };
        }
        await removeRecord(adapter, 'WAITING.json');
        return { applied: true };
      }
      default: {
        const _exhaustive: never = event;
        throw new Error(`Unknown SignalEvent type`);
      }
    }
  });
}

import type { NamedDocCategory, RootNamedDocKey } from 'get-shit-done/adapters/types.js';
import type { TransactionalAdapter } from './transaction.js';
import { getRecord, putRecord } from '../primitives/record.js';

function resolveNamedDocPath(
  category: NamedDocCategory,
  key: string,
  opts?: { workstream?: string },
): string {
  switch (category) {
    case 'root':         return `${key}.md`;  // HANDOFF.md etc.
    case 'research':     return `research/${key}.md`;
    case 'intel':        return `intel/${key}.md`;
    case 'codebase':     return `codebase/${key}.md`;
    case 'archived-milestone': {
      if (!opts?.workstream) {
        throw new Error(`putNamedDoc('archived-milestone', ...) requires opts.workstream (milestone version)`);
      }
      return `archived-milestones/${opts.workstream}/${key}.md`;
    }
    case 'reports':      return opts?.workstream ? `reports/${opts.workstream}/${key}.md` : `reports/${key}.md`;
    case 'sketches':     return `sketches/${key}.md`;
    case 'tmp':          return `tmp/${key}.md`;
    default: {
      const _exhaustive: never = category;
      throw new Error(`Unknown NamedDocCategory: ${String(_exhaustive)}`);
    }
  }
}

// Single implementation; BeadsAdapter declares both overloads in src/index.ts
export async function putNamedDoc(
  adapter: TransactionalAdapter,
  category: NamedDocCategory,
  key: string,
  body: string,
  opts?: { workstream?: string },
): Promise<void> {
  const path = resolveNamedDocPath(category, key, opts);
  await putRecord(adapter, path, body);
}

export async function getNamedDoc(
  adapter: TransactionalAdapter,
  category: NamedDocCategory,
  key: string,
  opts?: { workstream?: string },
): Promise<string | null> {
  const path = resolveNamedDocPath(category, key, opts);
  return await getRecord(adapter, path);
}
```
